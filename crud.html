#include using namespace std; int main() { int arr[10] = {20, 10, 40, 15, 25}; int sizea = 5; int add; int position; // OLD ARRAY cout << "OLD ARRAY: "; for (int i = 0; i < sizea; i++) { cout << arr[i] << " "; } // INPUT VALUE cout << "\nEnter value to add: "; cin >> add; // INPUT POSITION cout << "Enter position (0 to " << sizea << "): "; cin >> position; // CHECK SPACE AND POSITION if (sizea < 10 && position >= 0 && position <= sizea) { // SHIFT VALUES RIGHT for (int i = sizea; i > position; i--) { arr[i] = arr[i - 1]; } // ADD VALUE arr[position] = add; sizea++; // NEW ARRAY cout << "NEW ARRAY: "; for (int i = 0; i < sizea; i++) { cout << arr[i] << " "; } } else { cout << "Invalid position or array is full!"; } return 0; }
