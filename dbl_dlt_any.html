#include using namespace std; struct Quiz { int num; Quiz* prev; Quiz* next; }; int main() { // First Node Quiz* Q = new Quiz(); Q->num = 20; Q->prev = nullptr; Q->next = nullptr; // Second Node Quiz* Q1 = new Quiz(); Q1->num = 10; Q1->prev = nullptr; Q1->next = nullptr; // Third Node Quiz* Q2 = new Quiz(); Q2->num = 30; Q2->prev = nullptr; Q2->next = nullptr; // Connect Nodes Q->next = Q1; Q1->prev = Q; Q1->next = Q2; Q2->prev = Q1; // Deletion at Position 2 Quiz* temp = Q; for (int i = 1; i < 2; i++) { temp = temp->next; } // Delete Node temp->prev->next = temp->next; temp->next->prev = temp->prev; delete temp; // Display Linked List for (Quiz* temp = Q; temp != nullptr; temp = temp->next) { cout << temp->num << " <-> "; } cout << "NULL"; return 0; }
